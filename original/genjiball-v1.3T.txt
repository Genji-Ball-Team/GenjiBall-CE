disabled rule("Active game - check for last player surviving")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Is Game In Progress == True;
		Number Of Living Players(All Teams) == 1;
		Global.IsEnoughPlayersToStart == True;
		Global.duelMode == False;
	}

	actions
	{
		Global.IsInFinalDuel = False;
		Modify Player Score(All Living Players(All Teams), 1);
		Wait(0.250, Ignore Condition);
		If((Match Time == 0 || Global.TieBreakerActive) && !Global.tournamentmode);
			"If there is alredy a tiebreaker, then the tiebreaker has ended."
			disabled If(Global.TieBreakerActive);
			disabled Break;
			disabled End;
			Global.HighestScore = Score Of(Last Of(Sorted Array(All Players(All Teams), Score Of(Current Array Element))));
			"Check if there is more then 1 person with the highest score"
			If(Count Of(Filtered Array(All Players(All Teams), Score Of(Current Array Element) == Global.HighestScore)) > 1);
				"There is more then 1, activate tie breaker"
				Global.TieBreakerActive = True;
			Else;
				Global.TieBreakerActive = False;
				If(Match Time == 0);
					Enable Built-In Game Mode Music;
					Enable Built-In Game Mode Completion;
				End;
			End;
		End;
		Stop Chasing Global Variable(ballSpawnCountdown);
		Stop Chasing Player Variable(All Players(All Teams), bouncePadCooldown);
		Big Message(All Players(All Teams), Custom String("{0} has won the round", First Of(All Living Players(All Teams))));
		Wait(2, Ignore Condition);
		Skip If(Global.duelMode, 8);
		If(!Global.TieBreakerActive);
			Resurrect(All Dead Players(All Teams));
		Else;
			"Kill anyone alive who doesn't have the highest score."
			Kill(Filtered Array(All Living Players(All Teams), Score Of(Current Array Element) != Global.HighestScore), Null);
			"During tiebreaker, only revive the people who have the same score as the highest score."
			Resurrect(Filtered Array(All Dead Players(All Teams), Score Of(Current Array Element) == Global.HighestScore));
			Wait(0.250, Ignore Condition);
			Big Message(All Players(All Teams), Custom String("tiebreaker!"));
		End;
		Call Subroutine(startRound);
	}
}

rule("Collision - collision check")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Global.ballIsOut == True;
		Global.ballSpeed > 150;
	}

	actions
	{
		Wait Until(!Global.HasCollision, 99999);
		Global.HasCollision = True;
		If(Global.ballHitPosition != Null);
			"-o--x--------o->Between last ballPosition & middlePosition"
			Global.phasePosition1 = Global.ballHitPosition + Global.phasePosition;
			Global.phasePosition1 /= 2;
			"-o-----x-----o->Between last & current ballPosition"
			Global.phasePosition = Global.ballHitPosition;
			Global.ballHitPosition = Null;
		Else;
			"-o--x--------o->Between last ballPosition & middlePosition"
			Global.phasePosition1 = Global.phasePosition + Global.ballPosition;
			Global.phasePosition1 /= 2;
			Global.phasePosition1 += Global.phasePosition;
			Global.phasePosition1 /= 2;
			"-o-----x-----o->Between last & current ballPosition"
			Global.phasePosition += Global.ballPosition;
			Global.phasePosition /= 2;
		End;
		"-o--------x--o->Between phasePosition & current ballPosition & phasePosition"
		Global.phasePosition2 = Global.phasePosition + Global.ballPosition;
		Global.phasePosition2 /= 2;
		"-o-----------x->"
		If(Distance Between(Global.ballPosition, Eye Position(Global.Target)) <= 1.900);
			Call Subroutine(collisionTarget);
		"-o-----x-----o->"
		Else If(Distance Between(Global.phasePosition, Eye Position(Global.Target)) <= 1.900);
			Call Subroutine(collisionTarget);
		"-o--x--------o->"
		Else If(Distance Between(Global.phasePosition1, Eye Position(Global.Target)) <= 1.900);
			Call Subroutine(collisionTarget);
		"-o--------x--o->"
		Else If(Distance Between(Global.phasePosition2, Eye Position(Global.Target)) <= 1.900);
			Call Subroutine(collisionTarget);
		End;
		"Save last ballPosition for next loop"
		Global.phasePosition = Evaluate Once(Global.ballPosition);
		Global.HasCollision = False;
		Wait(0, Ignore Condition);
		Loop If Condition Is True;
	}
}

rule("Ball Physics - general collision")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Global.ballIsOut == True;
		Update Every Frame(Is In Line of Sight(Global.ballPosition, Global.ballPosition + Global.ballDirection * (
			Global.ballMotion != 2 ? Global.ballSpeed : 1) * 0.033, Barriers Do Not Block LOS)) == False;
		Y Component Of(Global.ballPosition) > 7.500;
		(Z Component Of(Global.ballPosition) > 20 || Z Component Of(Global.ballPosition) < -20 || X Component Of(Global.ballPosition)
			> 20 || X Component Of(Global.ballPosition) < -20 || Y Component Of(Global.ballPosition) > 0) == True;
	}

	actions
	{
		Global.ballNextFramePosition = Global.ballPosition + Global.ballDirection * Global.ballSpeed * 0.066;
		"Used for phasingprotection in Ball/Target - collision check"
		Global.ballHitPosition = Ray Cast Hit Position(Global.ballPosition, Global.ballNextFramePosition, Null, All Players(All Teams),
			False);
		If(Current Map == Map(Workshop Island) || Current Map == Map(Workshop Island Night) || (Y Component Of(Global.ballHitPosition)

				Global.ballHitPosition), Z Component Of(Global.ballHitPosition))), Distance Between(Global.ballHitPosition, Vector(-20,

				Global.ballHitPosition, Vector(X Component Of(Global.ballHitPosition), Y Component Of(Global.ballHitPosition), -20)));
			Call Subroutine(simpleIslandCollision);
		Else;
			Global.BallCollisionSurfaceNormal = Ray Cast Hit Normal(Global.ballPosition, Global.ballNextFramePosition, Null, All Players(
				All Teams), False);
			"new reflection formula --- new direction = v - 2 * (v ∙ n) * n"
			Global.ballDirection -= 2 * Dot Product(Global.ballDirection, Global.BallCollisionSurfaceNormal)
				* Global.BallCollisionSurfaceNormal;
		End;
		Play Effect(All Players(All Teams), Pharah Barrage Explosion Effect, All Teams, Global.ballHitPosition, 1);
		Play Effect(All Players(All Teams), Echo Sticky Bomb Explosion Sound, Color(White), Global.ballHitPosition, 30);

rule("Subroutine - simple island collision")
{
	event
	{
		Subroutine;
		simpleIslandCollision;
	}

	actions
	{
		Global.CollisionSurfaceIndex = Index Of Array Value(Global.CollisionSurfaceDistances, First Of(Sorted Array(
			Global.CollisionSurfaceDistances, Current Array Element)));
		If(Global.CollisionSurfaceIndex == 0);
			If(Y Component Of(Global.ballDirection) < 0);
				Global.ballDirection *= Vector(1, -1, 1);
			Else;
				Global.CollisionSurfaceIndex = Index Of Array Value(Global.CollisionSurfaceDistances, Sorted Array(
					Global.CollisionSurfaceDistances, Current Array Element)[1]);
			End;
		End;
		If(Global.CollisionSurfaceIndex == 1);
			If(X Component Of(Global.ballDirection) < 0);
				Global.ballDirection *= Vector(-1, 1, 1);
			End;
		Else If(Global.CollisionSurfaceIndex == 2);
			If(X Component Of(Global.ballDirection) > 0);
				Global.ballDirection *= Vector(-1, 1, 1);
			End;
		Else If(Global.CollisionSurfaceIndex == 3);
			If(Z Component Of(Global.ballDirection) < 0);
				Global.ballDirection *= Vector(1, 1, -1);
			End;
		Else If(Global.CollisionSurfaceIndex == 4);
			If(Z Component Of(Global.ballDirection) > 0);
				Global.ballDirection *= Vector(1, 1, -1);
			End;
		End;
	}
}

	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Current Game Mode == Game Mode(Team Deathmatch);
		Is Game In Progress == True;
		Global.RoundInProgress != False;
		Count Of(Filtered Array(All Living Players(Team 1), Current Array Element.hasMoved)) == 1;
		Count Of(Filtered Array(All Living Players(Team 2), Current Array Element.hasMoved)) == 1;
		Global.IsInFinalDuel == False;
		Global.IsEnoughPlayersToStart == True;
		Global.endlessMode == False;
	}

	actions
	{
		Wait(0.016, Abort When False);
		Abort If Condition Is False;
		Call Subroutine(setupFinal);
	}
}

rule("Teams - Last surviving")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Current Game Mode == Game Mode(Team Deathmatch);
		Is Game In Progress == True;
		Global.IsEnoughPlayersToStart == True;
		(Number Of Living Players(Team 1) == 0 || Number Of Living Players(Team 2) == 0) == True;
	}

	actions
	{
		Global.IsInFinalDuel = False;
		If(Number Of Living Players(Team 1) == 0);
			Modify Team Score(Team 2, 1);
		Else;
			Modify Team Score(Team 1, 1);
		End;
		Wait(0.250, Ignore Condition);
		If((Match Time == 0 || Global.TieBreakerActive) && !Global.tournamentmode);
			Global.HighestScore = Score Of(Last Of(Sorted Array(All Players(All Teams), Score Of(Current Array Element))));
			"Check if there is more then 1 person with the highest score"
			If(Team Score(Team 1) == Team Score(Team 2));
				"There is more then 1, activate tie breaker"
				Global.TieBreakerActive = True;
			Else;
				Global.TieBreakerActive = False;
				If(Match Time == 0);
					Enable Built-In Game Mode Music;
					Enable Built-In Game Mode Completion;
				End;
			End;
		End;
		Stop Chasing Global Variable(ballSpawnCountdown);
		Stop Chasing Player Variable(All Players(All Teams), bouncePadCooldown);
		Big Message(All Players(All Teams), Custom String("{0} has won the round", First Of(All Living Players(All Teams))));
		Wait(2, Ignore Condition);
		Skip If(Global.duelMode, 8);
		If(!Global.TieBreakerActive);
			Resurrect(All Dead Players(All Teams));
		Else;
			"Kill anyone alive who doesn't have the highest score."
			Kill(Filtered Array(All Living Players(All Teams), Score Of(Current Array Element) != Global.HighestScore), Null);
			"During tiebreaker, only revive the people who have the same score as the highest score."
			Resurrect(Filtered Array(All Dead Players(All Teams), Score Of(Current Array Element) == Global.HighestScore));
			Wait(0.250, Ignore Condition);
			Big Message(All Players(All Teams), Custom String("tiebreaker!"));
		End;
		Call Subroutine(startRound);
	}
}

rule("Sandbox - Set Ball spawnpoint")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Global.Sandbox == True;
		Is Dummy Bot(Event Player) != True;
		Is Game In Progress == True;
		Global.ballIsOut == False;
		Is Button Held(Event Player, Button(Interact)) == True;
		Global.Sandboxselected == 1;
	}

	actions
	{
		If(Global.Sandboxxyz == 1);
			If(Is Button Held(Event Player, Button(Ultimate)) == True);
				If(Global.Sandboxplusminus == 1);
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(1, 0, 0);
				Else;
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(-1, 0, 0);
				End;
			Else;
				If(Global.Sandboxplusminus == 1);
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0.250, 0, 0);
				Else;
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(-0.250, 0, 0);
				End;
			End;
		Else If(Global.Sandboxxyz == 2);
			If(Is Button Held(Event Player, Button(Ultimate)) == True);
				If(Global.Sandboxplusminus == 1);
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0, 1, 0);
				Else;
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0, -1, 0);
				End;
			Else;
				If(Global.Sandboxplusminus == 1);
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0, 0.250, 0);
				Else;
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0, -0.250, 0);
				End;
			End;
		Else If(Global.Sandboxxyz == 3);
			If(Is Button Held(Event Player, Button(Ultimate)) == True);
				If(Global.Sandboxplusminus == 1);
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0, 0, 1);
				Else;
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0, 0, -1);
				End;
			Else;
				If(Global.Sandboxplusminus == 1);
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0, 0, 0.250);
				Else;
					Global.Sandboxballpos = Global.Sandboxballpos + Vector(0, 0, -0.250);
				End;
			End;
		End;
	}
}

rule("Sandbox - Change xyz and +/-")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Global.Sandbox == True;
		Is Dummy Bot(Event Player) != True;
		Is Game In Progress == True;
		Global.ballIsOut == False;
		(Is Button Held(Event Player, Button(Jump)) == True || Is Button Held(Event Player, Button(Crouch)) == True) == True;
	}

	actions
	{
		Abort If Condition Is False;
		Wait(0.250, Ignore Condition);
		If(Is Button Held(Event Player, Button(Jump)) == True && Is Button Held(Event Player, Button(Crouch)) == True);
			Global.Sandboxplusminus = Global.Sandboxplusminus * -1;
			Big Message(All Players(All Teams), Custom String("+/- switched"));
		Else If(Is Button Held(Event Player, Button(Jump)) == True);
			If(Global.Sandboxxyz == 3);
				Big Message(All Players(All Teams), Custom String("Invalid input"));
			Else;
				Global.Sandboxxyz = Global.Sandboxxyz + 1;
				Big Message(All Players(All Teams), Custom String("axis +1"));
			End;
		Else If(Is Button Held(Event Player, Button(Crouch)) == True);
			If(Global.Sandboxxyz == 1);
				Big Message(All Players(All Teams), Custom String("Invalid input"));
			Else;
				Global.Sandboxxyz = Global.Sandboxxyz - 1;
				Big Message(All Players(All Teams), Custom String("axis -1"));
			End;
		End;
	}
}

rule("Sandbox - Switch between pos/dir/spd")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is Button Held(Event Player, Button(Reload)) == True;
		Is Dummy Bot(Event Player) != True;
		Is Game In Progress == True;
		Global.ballIsOut == False;
		Global.Sandbox == True;
	}

	actions
	{
		Wait Until(!Is Button Held(Event Player, Button(Reload)), 0.400);
		If(Is Button Held(Event Player, Button(Reload)) == True);
			If(Global.Sandboxselected > 2);
				Global.Sandboxselected = 1;
			Else;
				Global.Sandboxselected = Global.Sandboxselected + 1;
			End;
			Big Message(Event Player, Custom String("Switched pos/dir/spd to {0}", Global.Sandboxselected));
		End;
	}
}

disabled rule("Gb Abilities - stack crit slashes")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Event Player.Critalert == True;
		Global.Abilities == True;
		Is Dummy Bot(Event Player) != True;
	}

	actions
	{
		Event Player.Critalert = False;
		Wait(0.032, Restart When True);
		Global.ballSpeed = Global.ballSpeed * 1.750;
		Wait(5, Ignore Condition);
		Global.ballSpeed = Global.ballSpeed / 1.750;
		Global.ballSpeed = Global.ballSpeed * 1.250;
	}
