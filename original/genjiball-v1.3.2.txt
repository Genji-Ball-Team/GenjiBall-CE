rule("Active game - check for last player surviving")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Is Game In Progress == True;
		Number Of Living Players(All Teams) == 1;
		Global.IsEnoughPlayersToStart == True;
		Global.duelMode == False;
	}

	actions
	{
		Global.IsInFinalDuel = False;
		Modify Player Score(All Living Players(All Teams), 1);
		Wait(0.250, Ignore Condition);
		If((Match Time == 0 || Global.TieBreakerActive) && !Global.tournamentMode);
			"If there is alredy a tiebreaker, then the tiebreaker has ended.\nif TieBreakerActive:\nbreak\n__end__()"
			Global.HighestScore = Score Of(Last Of(Sorted Array(All Players(All Teams), Score Of(Current Array Element))));
			"Check if there is more than 1 person with the highest score"
			If(Count Of(Filtered Array(All Players(All Teams), Score Of(Current Array Element) == Global.HighestScore)) > 1);
				"There is more than 1, activate tie breaker"
				Global.TieBreakerActive = True;
			Else;
				Global.TieBreakerActive = False;
				If(Match Time == 0);
					Enable Built-In Game Mode Music;
					Enable Built-In Game Mode Completion;
				End;
			End;
		End;
		Stop Chasing Global Variable(ballSpawnCountdown);
		Stop Chasing Player Variable(All Players(All Teams), bouncePadCooldown);
		Big Message(All Players(All Teams), Custom String("{0} has won the round", First Of(All Living Players(All Teams))));
		Wait(2, Ignore Condition);
		Skip If(Global.duelMode, 8);
		If(!Global.TieBreakerActive);
			Resurrect(All Dead Players(All Teams));
		Else;
			"Kill anyone alive who doesn't have the highest score."
			Kill(Filtered Array(All Living Players(All Teams), Score Of(Current Array Element) != Global.HighestScore), Null);
			"During tiebreaker, only revive the people who have the same score as the highest score."
			Resurrect(Filtered Array(All Dead Players(All Teams), Score Of(Current Array Element) == Global.HighestScore));
			Wait(0.250, Ignore Condition);
			Big Message(All Players(All Teams), Custom String("tiebreaker!"));
		End;
		Call Subroutine(startRound);
	}
}

rule("Collision - collision check")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Global.ballIsOut == True;
		Global.ballSpeed > 150;
	}

	actions
	{
		If(Global.ballHitPosition != Null);
			"-o--x--------o->Between last ballPosition & middlePosition"
			Global.phasePosition1 = Global.ballHitPosition + Global.phasePosition;
			Global.phasePosition1 /= 2;
			"-o-----x-----o->Between last & current ballPosition"
			Global.phasePosition = Global.ballHitPosition;
			Global.ballHitPosition = Null;
		Else;
			"-o--x--------o->Between last ballPosition & middlePosition"
			Global.phasePosition1 = Global.phasePosition + Global.ballPosition;
			Global.phasePosition1 /= 2;
			Global.phasePosition1 += Global.phasePosition;
			Global.phasePosition1 /= 2;
			Global.phasePosition += Global.ballPosition;
			Global.phasePosition /= 2;
		End;
		"-o--------x--o->Between phasePosition & current ballPosition & phasePosition"
		Global.phasePosition2 = Global.phasePosition + Global.ballPosition;
		Global.phasePosition2 /= 2;
		"-o-----------x->"
		If(Distance Between(Global.ballPosition, Eye Position(Global.target)) <= 1.900);
			Call Subroutine(collisionTarget);
		Else If(Distance Between(Global.phasePosition, Eye Position(Global.target)) <= 1.900);
			Call Subroutine(collisionTarget);
		Else If(Distance Between(Global.phasePosition1, Eye Position(Global.target)) <= 1.900);
			Call Subroutine(collisionTarget);
		Else If(Distance Between(Global.phasePosition2, Eye Position(Global.target)) <= 1.900);
			Call Subroutine(collisionTarget);
		End;
		"Save last ballPosition for next loop"
		Global.phasePosition = Evaluate Once(Global.ballPosition);
		Wait(0, Ignore Condition);
		Loop If Condition Is True;
	}
}

rule("Ball Physics - anti ghost correction")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Global.antiGhostMode != 0;
		Global.ballIsOut == True;
		Global.ballMotion == 0;
		Global.target != Null;
		Is Alive(Global.target) == True;
	}

	actions
	{
		"Mode 2 = steer. If the ball is close to the target and its current\ndirection is far off from the ideal target direction, temporarily raise\nsteering strength. Experimental leaves AntiGhost off by default;\nTournament+ uses Steer."
		If(Distance Between(Global.ballPosition, Eye Position(Global.target)) <= 7);
			If(Global.antiGhostMode == 1);
				If(Global.antiGhostPrevDistance == 999);
					Global.antiGhostPrevDistance = Distance Between(Global.ballPosition, Eye Position(Global.target));
				End;
				If(Distance Between(Global.ballPosition, Eye Position(Global.target)) >= Global.antiGhostPrevDistance - 0.050);
					Global.antiGhostStallTimer += 0.050;
				Else;
					Global.antiGhostStallTimer = 0;
				End;
				Global.antiGhostPrevDistance = Distance Between(Global.ballPosition, Eye Position(Global.target));
				If(Global.antiGhostStallTimer >= 0.150);
					If(Global.ballDirectionRate < 4);
						Global.ballDirectionRate = 4;
					End;
				Else If(Global.ballDirectionRate == 4);
					Global.ballDirectionRate = Global.ballDirectionRateInit;
				End;
			Else If(Global.antiGhostMode == 2);
				If(Angle Between Vectors(Global.ballDirection, Direction Towards(Global.ballPosition, Eye Position(Global.target))) >= 35);
					Global.antiGhostStallTimer = 0;
					Global.antiGhostPrevDistance = Distance Between(Global.ballPosition, Eye Position(Global.target));
					If(Global.ballDirectionRate < 4);
						Global.ballDirectionRate = 4;
					End;
				Else If(Global.ballDirectionRate == 4);
					Global.ballDirectionRate = Global.ballDirectionRateInit;
				End;
			End;
		Else;
			Global.antiGhostPrevDistance = 999;
			Global.antiGhostStallTimer = 0;
			If(Global.ballDirectionRate == 4);
				Global.ballDirectionRate = Global.ballDirectionRateInit;
			End;
		End;
		Wait(0.050, Abort When False);
		Loop If Condition Is True;
	}
}

rule("Subroutine - simple island collision")
{
	event
	{
		Subroutine;
		simpleIslandCollision;
	}

	actions
	{
		Global.CollisionSurfaceIndex = Index Of Array Value(Global.CollisionSurfaceDistances, First Of(Sorted Array(
			Global.CollisionSurfaceDistances, Current Array Element)));
		If(Global.CollisionSurfaceIndex == 0);
			If(Y Component Of(Global.ballDirection) < 0);
				Global.ballDirection *= Vector(1, -1, 1);
			Else;
				Global.CollisionSurfaceIndex = Index Of Array Value(Global.CollisionSurfaceDistances, Sorted Array(
					Global.CollisionSurfaceDistances, Current Array Element)[1]);
			End;
		End;
		If(Global.CollisionSurfaceIndex == 1);
			If(X Component Of(Global.ballDirection) < 0);
				Global.ballDirection *= Vector(-1, 1, 1);
			End;
		Else If(Global.CollisionSurfaceIndex == 2);
			If(X Component Of(Global.ballDirection) > 0);
				Global.ballDirection *= Vector(-1, 1, 1);
			End;
		Else If(Global.CollisionSurfaceIndex == 3);
			If(Z Component Of(Global.ballDirection) < 0);
				Global.ballDirection *= Vector(1, 1, -1);
			End;
		Else If(Global.CollisionSurfaceIndex == 4);
			If(Z Component Of(Global.ballDirection) > 0);
				Global.ballDirection *= Vector(1, 1, -1);
			End;
		End;
	}
}

rule("zBozo - face target while deflecting")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Global.bozo_Bot == True;
		Distance Between(Eye Position(Event Player), Position Of(Global.ballPosition)) <= 2 * (Global.ballSpeed / 20);
		Global.target == Event Player;
		Is Dummy Bot(Event Player) == True;
		Global.ballIsOut == True;
		Is Alive(Event Player) == True;
		Is Using Ability 1(Event Player) == False;
		Is Using Ability 2(Event Player) == False;
		Has Status(Event Player, Stunned) == False;
		Global.bozoIsGhostDashing != True;
	}

	actions
	{
		If(Event Player.canDash == True);
			If(Random Integer(0, 8) == 0);
				If(Global.ballSpeed < 80);
					Start Facing(Event Player, Direction From Angles(Horizontal Angle From Direction(Direction Towards(Eye Position(Event Player),
						Eye Position(Evaluate Once(Random Value In Array(Remove From Array(All Living Players(All Teams), Event Player))))))
						+ Evaluate Once(Random Integer(-35, 35)), Evaluate Once(Random Integer(-35, 35))), 10000, To World, Direction and Turn Rate);
				Else;
					Start Facing(Event Player, Direction From Angles(Horizontal Angle From Direction(Direction Towards(Eye Position(Event Player),
						Eye Position(Evaluate Once(Random Value In Array(Remove From Array(All Living Players(All Teams), Event Player))))))
						+ Evaluate Once(Random Integer(-10, 10)), Evaluate Once(Random Integer(-10, 10))), 10000, To World, Direction and Turn Rate);
				End;
			Else;
				Start Facing(Event Player, Vector(Random Integer(-360, 360), Random Integer(-360, 360), Random Integer(-360, 360)), 10000,
					To World, Direction and Turn Rate);
			End;
		Else;
			If(Random Integer(0, 1) == 0);
				Start Facing(Event Player, Direction From Angles(Horizontal Angle From Direction(Direction Towards(Eye Position(Event Player),
					Eye Position(Player Closest To Reticle(Event Player, All Teams)))) + Evaluate Once(Random Integer(-90, -45)), Evaluate Once(
					Random Integer(-90, 90))), 10000, To World, Direction and Turn Rate);
			Else;
				Start Facing(Event Player, Direction From Angles(Horizontal Angle From Direction(Direction Towards(Eye Position(Event Player),
					Eye Position(Player Closest To Reticle(Event Player, All Teams)))) + Evaluate Once(Random Integer(45, 90)), Evaluate Once(
					Random Integer(-90, 90))), 10000, To World, Direction and Turn Rate);
			End;
			Wait(0, Ignore Condition);
			Loop If(Is True For Any(Remove From Array(All Living Players(All Teams), Event Player), Is In View Angle(Event Player,
				Eye Position(Current Array Element), 30)));
		End;
		Wait Until(Global.target != Event Player || Is Dead(Event Player), 99999);
		If(Angle Between Vectors(Global.ballDirection * -1, Direction Towards(Global.target, Global.ballPosition)) < 20);
			"if distance(eventPlayer.getEyePosition(), target.getEyePosition()) < ballSpeed * 0.3:\nbozoWillEdash = true\nbigMessage(getAllPlayers(), \"EDASH\")"
			If(Distance Between(Eye Position(Event Player), Global.ballPosition) < Global.ballSpeed);
				Global.bozoWillGhostDash = True;
			End;
		End;
		Wait(0.300, Ignore Condition);
		Loop If Condition Is True;
	}
}

disabled rule("Teams - Last surviving")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Current Game Mode == Game Mode(Team Deathmatch);
		Is Game In Progress == True;
		Global.IsEnoughPlayersToStart == True;
		(Number Of Living Players(Team 1) == 0 || Number Of Living Players(Team 2) == 0) == True;
	}

	actions
	{
		Global.IsInFinalDuel = False;
		If(Number Of Living Players(Team 1) == 0);
			Modify Team Score(Team 2, 1);
		Else;
			Modify Team Score(Team 1, 1);
		End;
		Wait(0.250, Ignore Condition);
		If((Match Time == 0 || Global.TieBreakerActive) && !Global.tournamentMode);
			Global.HighestScore = Score Of(Last Of(Sorted Array(All Players(All Teams), Score Of(Current Array Element))));
			"Check if there is more than 1 person with the highest score"
			If(Team Score(Team 1) == Team Score(Team 2));
				"There is more than 1, activate tie breaker"
				Global.TieBreakerActive = True;
			Else;
				Global.TieBreakerActive = False;
				If(Match Time == 0);
					Enable Built-In Game Mode Music;
					Enable Built-In Game Mode Completion;
				End;
			End;
		End;
		Stop Chasing Global Variable(ballSpawnCountdown);
		Stop Chasing Player Variable(All Players(All Teams), bouncePadCooldown);
		Big Message(All Players(All Teams), Custom String("{0} has won the round", First Of(All Living Players(All Teams))));
		Wait(2, Ignore Condition);
		Skip If(Global.duelMode, 8);
		If(!Global.TieBreakerActive);
			Resurrect(All Dead Players(All Teams));
		Else;
			"Kill anyone alive who doesn't have the highest score."
			Kill(Filtered Array(All Living Players(All Teams), Score Of(Current Array Element) != Global.HighestScore), Null);
			"During tiebreaker, only revive the people who have the same score as the highest score."
			Resurrect(Filtered Array(All Dead Players(All Teams), Score Of(Current Array Element) == Global.HighestScore));
			Wait(0.250, Ignore Condition);
			Big Message(All Players(All Teams), Custom String("tiebreaker!"));
		End;
		Call Subroutine(startRound);
	}
