rule ("Initialization - Set map") {
    event {
        Ongoing - Global;
    }
    actions {
        "They are smaller than the others: lower the max players there.\nRule names can't contain \"Blizzard\", so the map is written \"Blizz World\" in rule names.\nAny other map has no row and no arena, so no round starts there (MAP_UNSUPPORTED)."
        Set Global Variable(mapRow, Value In Array(Array(Array(Vector(0, 1.199, 0), 60), Array(Vector(0, 1.199, 0), 60), Array(Vector(0, 1.199, 0), 29.7), Array(Vector(0, 1.199, 0), 50), Array(Vector(0, 1.199, 0), 50), Array(Vector(0, 1.199, 0), 60)), Index Of Array Value(Array(Map(Workshop Island), Map(Workshop Island Night), Map(Workshop Chamber), Map(Workshop Expanse), Map(Workshop Expanse Night), Map(Workshop Green Screen)), Current Map)));
        Set Global Variable(circleCenter, First Of(Global.mapRow));
        Set Global Variable(SphereSize, Value In Array(Global.mapRow, 1));
        "Kept large enough for the final duel spawns: 15 m out on Workshop Island (in bounds needs at\nleast 10) and SphereSize / 1.5 elsewhere (must stay outside the center exclusion's push zone)."
        If(Compare(Value In Array(Global.arenaSettings, 1), >, 0));
            Set Global Variable(SphereSize, Max(Value In Array(Global.arenaSettings, 1), Max(10, Multiply(1.5, Add(First Of(Global.arenaSettings), 0.5)))));
        End;
        "Workshop Island has water and a ledge instead of a wall; several rules check this."
        Set Global Variable(isIsland, Or(Compare(Current Map, ==, Map(Workshop Island)), Compare(Current Map, ==, Map(Workshop Island Night))));
    }
}

rule ("Active game - check for last player surviving") {
    event {
        Ongoing - Global;
    }
    conditions {
        Is Game In Progress == True;
        Number Of Living Players(All Teams) == 1;
        Global.IsEnoughPlayersToStart == True;
        Value In Array(Global.addOnSettings, 5) == False;
    }
    actions {
        Set Global Variable(IsInFinalDuel, False);
        "Cleared by startRound once the next round is under way."
        Set Global Variable(RoundSetupPending, True);
        Modify Player Score(All Living Players(All Teams), 1);
        Wait(0.25, Ignore Condition);
        If(And(Or(Compare(Match Time, ==, 0), Global.TieBreakerActive), Not(Global.tournamentMode)));
            "If there is alredy a tiebreaker, then the tiebreaker has ended.\nif TieBreakerActive:\nbreak\n__end__()"
            Set Global Variable(HighestScore, Score Of(Last Of(Sorted Array(All Players(All Teams), Score Of(Current Array Element)))));
            "Check if there is more than 1 person with the highest score"
            If(Compare(Count Of(Filtered Array(All Players(All Teams), Compare(Score Of(Current Array Element), ==, Global.HighestScore))), >, 1));
                "There is more than 1, activate tie breaker"
                Set Global Variable(TieBreakerActive, True);
            Else;
                Set Global Variable(TieBreakerActive, False);
                If(Compare(Match Time, ==, 0));
                    Enable Built-In Game Mode Music;
                    Enable Built-In Game Mode Completion;
                End;
            End;
        End;
        Stop Chasing Global Variable(ballSpawnCountdown);
        Stop Chasing Player Variable(All Players(All Teams), bouncePadCooldown);
        Big Message(All Players(All Teams), Custom String("{0} has won the round", First Of(All Living Players(All Teams))));
        Wait(Value In Array(Global.roundTimings, 1), Ignore Condition);
        Skip If(Value In Array(Global.addOnSettings, 5), 8);
        If(Not(Global.TieBreakerActive));
            Resurrect(All Dead Players(All Teams));
        Else;
            "Kill anyone alive who doesn't have the highest score."
            Kill(Filtered Array(All Living Players(All Teams), Compare(Score Of(Current Array Element), !=, Global.HighestScore)), Null);
            "During tiebreaker, only revive the people who have the same score as the highest score."
            Resurrect(Filtered Array(All Dead Players(All Teams), Compare(Score Of(Current Array Element), ==, Global.HighestScore)));
            Wait(0.25, Ignore Condition);
            Big Message(All Players(All Teams), Custom String("tiebreaker!"));
        End;
        //lbl_0:
        Call Subroutine(startRound);
    }
}

rule ("Collision - collision check") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.ballIsOut == True;
        "\"reliable hit detection\" (off by default, ball feel) runs this every tick at any speed, as a backup for\n\"Collision - ball reaches player\", whose condition reads the chased ballPosition."
        Or(Compare(Global.ballSpeed, >, Value In Array(Global.ballFeel, 9)), Value In Array(Global.ballFeel, 10)) == True;
    }
    actions {
        If(Compare(Global.ballHitPosition, !=, Null));
            "-o--x--------o->Between last ballPosition & middlePosition"
            Set Global Variable At Index(phasePosition, 1, Add(Global.ballHitPosition, First Of(Global.phasePosition)));
            Modify Global Variable At Index(phasePosition, 1, Divide, 2);
            "-o-----x-----o->Between last & current ballPosition"
            Set Global Variable At Index(phasePosition, 0, Global.ballHitPosition);
            Set Global Variable(ballHitPosition, Null);
        Else;
            "-o--x--------o->Between last ballPosition & middlePosition"
            Set Global Variable At Index(phasePosition, 1, Add(First Of(Global.phasePosition), Global.ballPosition));
            Modify Global Variable At Index(phasePosition, 1, Divide, 2);
            Modify Global Variable At Index(phasePosition, 1, Add, First Of(Global.phasePosition));
            Modify Global Variable At Index(phasePosition, 1, Divide, 2);
            Modify Global Variable At Index(phasePosition, 0, Add, Global.ballPosition);
            Modify Global Variable At Index(phasePosition, 0, Divide, 2);
        End;
        "-o-----------x->"
        If(Compare(Distance Between(Global.ballPosition, Eye Position(Global.target)), <=, Value In Array(Global.ballFeel, 2)));
            Call Subroutine(collisionTarget);
        Else If(Compare(Distance Between(First Of(Global.phasePosition), Eye Position(Global.target)), <=, Value In Array(Global.ballFeel, 2)));
            Call Subroutine(collisionTarget);
        Else If(Compare(Distance Between(Value In Array(Global.phasePosition, 1), Eye Position(Global.target)), <=, Value In Array(Global.ballFeel, 2)));
            Call Subroutine(collisionTarget);
        Else If(Compare(Distance Between(Divide(Add(First Of(Global.phasePosition), Global.ballPosition), 2), Eye Position(Global.target)), <=, Value In Array(Global.ballFeel, 2)));
            Call Subroutine(collisionTarget);
        End;
        "Save last ballPosition for next loop"
        Set Global Variable At Index(phasePosition, 0, Evaluate Once(Global.ballPosition));
        Wait(0, Ignore Condition);
        Loop If Condition Is True;
    }
}
