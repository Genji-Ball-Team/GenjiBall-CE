    }

        Is Game In Progress == True;

        Global.duelMode == False;
    }
    actions {
        Set Global Variable(IsInFinalDuel, False);
        Modify Player Score(All Living Players(All Teams), 1);
        Wait(0.25, Ignore Condition);
        If(And(Or(Compare(Match Time, ==, 0), Global.TieBreakerActive), Not(Global.tournamentmode)));
            "If there is alredy a tiebreaker, then the tiebreaker has ended.\nif TieBreakerActive:\nbreak\n__end__()"
            Set Global Variable(HighestScore, Score Of(Last Of(Sorted Array(All Players(All Teams), Score Of(Current Array Element)))));
            "Check if there is more then 1 person with the highest score"
            If(Compare(Count Of(Filtered Array(All Players(All Teams), Compare(Score Of(Current Array Element), ==, Global.HighestScore))), >, 1));
                "There is more then 1, activate tie breaker"
                Set Global Variable(TieBreakerActive, True);
            Else;
                Set Global Variable(TieBreakerActive, False);
                If(Compare(Match Time, ==, 0));
                    Enable Built-In Game Mode Music;
                    Enable Built-In Game Mode Completion;
                End;
            End;
        End;
        Stop Chasing Global Variable(ballSpawnCountdown);
        Stop Chasing Player Variable(All Players(All Teams), bouncePadCooldown);
        Big Message(All Players(All Teams), Custom String("{0} has won the round", First Of(All Living Players(All Teams))));
        Wait(2, Ignore Condition);
        Skip If(Global.duelMode, 8);
        If(Not(Global.TieBreakerActive));
            Resurrect(All Dead Players(All Teams));
        Else;
            "Kill anyone alive who doesn't have the highest score."
            Kill(Filtered Array(All Living Players(All Teams), Compare(Score Of(Current Array Element), !=, Global.HighestScore)), Null);
            "During tiebreaker, only revive the people who have the same score as the highest score."
            Resurrect(Filtered Array(All Dead Players(All Teams), Compare(Score Of(Current Array Element), ==, Global.HighestScore)));
            Wait(0.25, Ignore Condition);
            Big Message(All Players(All Teams), Custom String("tiebreaker!"));
        End;
        //lbl_0:
        Call Subroutine(startRound);
    }
}

rule ("Collision - collision check") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.ballIsOut == True;
        Global.ballSpeed > 150;
    }
    actions {
        Wait Until(Not(Global.HasCollision), 99999);
        Set Global Variable(HasCollision, True);
        If(Compare(Global.ballHitPosition, !=, Null));
            "-o--x--------o->Between last ballPosition & middlePosition"
            Set Global Variable(phasePosition1, Add(Global.ballHitPosition, Global.phasePosition));
            Modify Global Variable(phasePosition1, Divide, 2);
            "-o-----x-----o->Between last & current ballPosition"
            Set Global Variable(phasePosition, Global.ballHitPosition);
            Set Global Variable(ballHitPosition, Null);
        Else;
            "-o--x--------o->Between last ballPosition & middlePosition"
            Set Global Variable(phasePosition1, Add(Global.phasePosition, Global.ballPosition));
            Modify Global Variable(phasePosition1, Divide, 2);
            Modify Global Variable(phasePosition1, Add, Global.phasePosition);
            Modify Global Variable(phasePosition1, Divide, 2);
            Modify Global Variable(phasePosition, Add, Global.ballPosition);
            Modify Global Variable(phasePosition, Divide, 2);
        End;
        "-o--------x--o->Between phasePosition & current ballPosition & phasePosition"
        Set Global Variable(phasePosition2, Add(Global.phasePosition, Global.ballPosition));
        Modify Global Variable(phasePosition2, Divide, 2);
        "-o-----------x->"
        If(Compare(Distance Between(Global.ballPosition, Eye Position(Global.Target)), <=, 1.9));
            Call Subroutine(collisionTarget);
        "-o-----x-----o->"
        Else If(Compare(Distance Between(Global.phasePosition, Eye Position(Global.Target)), <=, 1.9));
            Call Subroutine(collisionTarget);
        "-o--x--------o->"
        Else If(Compare(Distance Between(Global.phasePosition1, Eye Position(Global.Target)), <=, 1.9));
            Call Subroutine(collisionTarget);
        "-o--------x--o->"
        Else If(Compare(Distance Between(Global.phasePosition2, Eye Position(Global.Target)), <=, 1.9));
            Call Subroutine(collisionTarget);
        End;
        "Save last ballPosition for next loop"
        Set Global Variable(phasePosition, Evaluate Once(Global.ballPosition));
        Set Global Variable(HasCollision, False);
        Wait(0, Ignore Condition);
        Loop If Condition Is True;
    }
}

rule ("Subroutine - simple island collision") {
    event {
        Subroutine;
        simpleIslandCollision;
    }
    actions {
        Set Global Variable(CollisionSurfaceIndex, Index Of Array Value(Global.CollisionSurfaceDistances, First Of(Sorted Array(Global.CollisionSurfaceDistances, Current Array Element))));
        If(Compare(Global.CollisionSurfaceIndex, ==, 0));
            If(Compare(Y Component Of(Global.ballDirection), <, 0));
                Modify Global Variable(ballDirection, Multiply, Vector(1, -1, 1));
            Else;
                Set Global Variable(CollisionSurfaceIndex, Index Of Array Value(Global.CollisionSurfaceDistances, Value In Array(Sorted Array(Global.CollisionSurfaceDistances, Current Array Element), 1)));
            End;
        End;
        If(Compare(Global.CollisionSurfaceIndex, ==, 1));
            If(Compare(X Component Of(Global.ballDirection), <, 0));
                Modify Global Variable(ballDirection, Multiply, Vector(-1, 1, 1));
            End;
        Else If(Compare(Global.CollisionSurfaceIndex, ==, 2));
            If(Compare(X Component Of(Global.ballDirection), >, 0));
                Modify Global Variable(ballDirection, Multiply, Vector(-1, 1, 1));
            End;
        Else If(Compare(Global.CollisionSurfaceIndex, ==, 3));
            If(Compare(Z Component Of(Global.ballDirection), <, 0));
                Modify Global Variable(ballDirection, Multiply, Vector(1, 1, -1));
            End;
        Else If(Compare(Global.CollisionSurfaceIndex, ==, 4));
            If(Compare(Z Component Of(Global.ballDirection), >, 0));
                Modify Global Variable(ballDirection, Multiply, Vector(1, 1, -1));
            End;
        End;
    }
}

rule ("zBozo - face when Deflecting") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Global.bozo_Bot == True;
        Distance Between(Eye Position(Event Player), Position Of(Global.ballPosition)) <= Multiply(2, Divide(Global.ballSpeed, 20));
        Global.Target == Event Player;
        Is Dummy Bot(Event Player) == True;
        Global.ballIsOut == True;

        Is Using Ability 1(Event Player) == False;
        Is Using Ability 2(Event Player) == False;
        Has Status(Event Player, Stunned) == False;
        Global.bozoIsGhostDashing != True;
    }
    actions {
        If(Compare((Event Player).canDash, ==, True));
            If(Compare(Random Integer(0, 8), ==, 0));
                If(Compare(Global.ballSpeed, <, 80));
                    Start Facing(Event Player, Direction From Angles(Add(Horizontal Angle From Direction(Direction Towards(Eye Position(Event Player), Eye Position(Evaluate Once(Random Value In Array(Remove From Array(All Living Players(All Teams), Event Player)))))), Evaluate Once(Random Integer(-35, 35))), Evaluate Once(Random Integer(-35, 35))), 10000, To World, Direction and Turn Rate);
                Else;
                    Start Facing(Event Player, Direction From Angles(Add(Horizontal Angle From Direction(Direction Towards(Eye Position(Event Player), Eye Position(Evaluate Once(Random Value In Array(Remove From Array(All Living Players(All Teams), Event Player)))))), Evaluate Once(Random Integer(-10, 10))), Evaluate Once(Random Integer(-10, 10))), 10000, To World, Direction and Turn Rate);
                End;
            Else;
                Start Facing(Event Player, Vector(Random Integer(-360, 360), Random Integer(-360, 360), Random Integer(-360, 360)), 10000, To World, Direction and Turn Rate);
            End;
        Else;
            If(Compare(Random Integer(0, 1), ==, 0));
                Start Facing(Event Player, Direction From Angles(Add(Horizontal Angle From Direction(Direction Towards(Eye Position(Event Player), Eye Position(Player Closest To Reticle(Event Player, All Teams)))), Evaluate Once(Random Integer(-90, -45))), Evaluate Once(Random Integer(-90, 90))), 10000, To World, Direction and Turn Rate);
            Else;
                Start Facing(Event Player, Direction From Angles(Add(Horizontal Angle From Direction(Direction Towards(Eye Position(Event Player), Eye Position(Player Closest To Reticle(Event Player, All Teams)))), Evaluate Once(Random Integer(45, 90))), Evaluate Once(Random Integer(-90, 90))), 10000, To World, Direction and Turn Rate);
            End;
            Wait(0, Ignore Condition);
            Loop If(Is True For Any(Remove From Array(All Living Players(All Teams), Event Player), Is In View Angle(Event Player, Eye Position(Current Array Element), 30)));
        End;
        Wait Until(Or(Compare(Global.Target, !=, Event Player), Is Dead(Event Player)), 99999);
        If(Compare(Angle Between Vectors(Multiply(Global.ballDirection, -1), Direction Towards(Global.Target, Global.ballPosition)), <, 20));
            "if distance(eventPlayer.getEyePosition(), Target.getEyePosition()) < ballSpeed * 0.3:\nbozoWillEdash = true\nbigMessage(getAllPlayers(), \"EDASH\")"
            If(Compare(Distance Between(Eye Position(Event Player), Global.ballPosition), <, Global.ballSpeed));
                Set Global Variable(bozoWillGhostDash, True);
            End;
        End;
        Wait(0.3, Ignore Condition);
        Loop If Condition Is True;
    }
}

        Global.IsEnoughPlayersToStart == True;
        Or(Compare(Number Of Living Players(Team 1), ==, 0), Compare(Number Of Living Players(Team 2), ==, 0)) == True;
    }
    actions {
        Set Global Variable(IsInFinalDuel, False);
        If(Compare(Number Of Living Players(Team 1), ==, 0));
            Modify Team Score(Team 2, 1);
        Else;
            Modify Team Score(Team 1, 1);
        End;
        Wait(0.25, Ignore Condition);
        If(And(Or(Compare(Match Time, ==, 0), Global.TieBreakerActive), Not(Global.tournamentmode)));
            Set Global Variable(HighestScore, Score Of(Last Of(Sorted Array(All Players(All Teams), Score Of(Current Array Element)))));
            "Check if there is more then 1 person with the highest score"
            If(Compare(Team Score(Team 1), ==, Team Score(Team 2)));
                "There is more then 1, activate tie breaker"
                Set Global Variable(TieBreakerActive, True);
            Else;
                Set Global Variable(TieBreakerActive, False);
                If(Compare(Match Time, ==, 0));
                    Enable Built-In Game Mode Music;
                    Enable Built-In Game Mode Completion;
                End;
            End;
        End;
        Stop Chasing Global Variable(ballSpawnCountdown);
        Stop Chasing Player Variable(All Players(All Teams), bouncePadCooldown);
        Big Message(All Players(All Teams), Custom String("{0} has won the round", First Of(All Living Players(All Teams))));
        Wait(2, Ignore Condition);
        Skip If(Global.duelMode, 8);
        If(Not(Global.TieBreakerActive));
            Resurrect(All Dead Players(All Teams));
        Else;
            "Kill anyone alive who doesn't have the highest score."
            Kill(Filtered Array(All Living Players(All Teams), Compare(Score Of(Current Array Element), !=, Global.HighestScore)), Null);
            "During tiebreaker, only revive the people who have the same score as the highest score."
            Resurrect(Filtered Array(All Dead Players(All Teams), Compare(Score Of(Current Array Element), ==, Global.HighestScore)));
            Wait(0.25, Ignore Condition);
            Big Message(All Players(All Teams), Custom String("tiebreaker!"));
        End;
        //lbl_0:
        Call Subroutine(startRound);
    }
