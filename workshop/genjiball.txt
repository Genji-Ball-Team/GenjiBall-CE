rule ("Settings - Workshop settings") {
    event {
        Ongoing - Global;
    }
    actions {
        "00 - Preset: Default for a ranked match, Tournament for a tourney (the host tool's tourney code)"
        Set Global Variable(presetMode, Workshop Setting Combo(Custom String("00 - Preset"), Custom String("preset"), 0, Array(Custom String("Default"), Custom String("Tournament")), 0));
        "One entry per AddOnSetting field (config/constants.opy), in that order.\nEach setting may only be read once, so read them here and nowhere else.\nDouble sens is on by default in ranked: hosts and players asked for it there."
        Set Global Variable(addOnSettings, Array(Workshop Setting Toggle(Custom String("30 - Player"), Custom String("double sens"), True, 7), Workshop Setting Integer(Custom String("30 - Player"), Custom String("double sens %"), 275, 100, 500, 8), Workshop Setting Toggle(Custom String("80 - Visual"), Custom String("red-green colorblind filter"), False, 4), Workshop Setting Toggle(Custom String("90 - Debug"), Custom String("debug HUD"), False, 0)));
        "70 - Ranked: one entry per RankedField (config/constants.opy), in that order.\nLOGGING to DEBUG_HUD are settings; the rest is set by features/ranked-log.opy."
        Set Global Variable(rankedState, Array(Workshop Setting Toggle(Custom String("70 - Ranked"), Custom String("ranked logging"), True, 0), Workshop Setting Toggle(Custom String("70 - Ranked"), Custom String("rank tags"), True, 1), Workshop Setting Real(Custom String("70 - Ranked"), Custom String("rank tag height"), 1.5, 0, 5, 2), Workshop Setting Real(Custom String("70 - Ranked"), Custom String("rank tag size"), 1, 0.5, 4, 3), Workshop Setting Toggle(Custom String("70 - Ranked"), Custom String("top 10 list"), True, 4), Workshop Setting Toggle(Custom String("70 - Ranked"), Custom String("ranked debug HUD"), False, 5), False));
        "The ruleset, the same for both presets: Tournament only adds tournament mode (round limit, breaks,\nno dashing into the water, basic anti-orbit). A tourney code turns it on too (features/tourney.opy)."
        Set Global Variable(tournamentMode, Compare(Global.presetMode, ==, 1));
        "Match: 15 min, or 30 rounds with a 60 s break every 10 in tournament mode"
        Set Global Variable(matchTime, 900);
        Set Global Variable(totalRounds, 30);
        Set Global Variable(breaksEnabled, True);
        Set Global Variable(roundsUntilBreak, 10);
        Set Global Variable(breakLength, 60);
        Set Global Variable(setballspawncountdownoriginal, 5);
        "One entry per RoundTiming field (config/constants.opy), in that order: ball respawn delay, round win pause"
        Set Global Variable(roundTimings, Array(2, 2));
        "Ball: start speed, max speed, speed gained per second"
        Set Global Variable(ballSpawnSpeed, 60);
        Set Global Variable(ballSpeedCap, 400);
        Set Global Variable(ballAcceleration, 0.25);
        Set Global Variable(ballFeel, Array(False, 1.05, 1.9, 1.75, 5, 6, 0.05, 0.3, 0.5, 150, False));
        "Moderate water: the ball skims off it at y = -15"
        Set Global Variable(waterLevel, -15);
        "Agile mobility: move speed, gravity and jump in %. Gravity stays above 0: currentGravity == 0 means\n\"in the water\" (maps/restrictions.opy)."
        Set Global Variable(moveSpeed, 190);
        Set Global Variable(gravity, 80);
        Set Global Variable(jumpVerticalSpeed, 180);
        "One entry per PlayerSetting field (config/constants.opy), in that order: dash cooldown, dash hit knockdown and its cooldown"
        Set Global Variable(playerSettings, Array(3.04, 1, 0.6));
    }
}

rule ("Tourney - settings") {
    event {
        Ongoing - Global;
    }
    conditions {
        First Of(Value In Array(Global.rankedState, 16)) == True;
    }
    actions {
        "No match timer and no timed tiebreaker, whatever the preset. Anything but Tournament is unranked anyway."
        Set Global Variable(tournamentMode, True);
        "The rule's round limit replaces \"tournament rounds\" and the preset's value"
        Set Global Variable(totalRounds, If-Then-Else(Compare(Value In Array(Value In Array(Global.rankedState, 16), 2), >=, 1), Round To Integer(Value In Array(Value In Array(Global.rankedState, 16), 2), Down), 30));
        "The ranked log counts the rounds and ends the match, and the server needs the log"
        Set Global Variable At Index(rankedState, 0, True);
    }
}

rule ("Active game - check for last player surviving") {
    event {
        Ongoing - Global;
    }
    conditions {
        Is Game In Progress == True;
        Number Of Living Players(All Teams) == 1;
        Global.IsEnoughPlayersToStart == True;
    }
    actions {
        Set Global Variable(IsInFinalDuel, False);
        "Cleared by startRound once the next round is under way."
        Set Global Variable(RoundSetupPending, True);
        Modify Player Score(All Living Players(All Teams), 1);
        Wait(0.25, Ignore Condition);
        "The host's sit-out takes effect between rounds, after the ranked log's ROUND_END. With too few players\nleft, \"Check for <2 players\" stops the lobby during the win pause"
        Call Subroutine(applyHostSitOut);
        "The tie is decided after the win pause. Until then, a tie with the host playing or sitting out stops \"all players\ndead\" and \"end game\" from ending the match during the pause (the winner leaves). Never ends a match or tiebreaker"
        If(And(Or(Compare(Match Time, ==, 0), Global.TieBreakerActive), Not(Global.tournamentMode)));
            Set Global Variable(HighestScore, Score Of(Last Of(Sorted Array(All Players(All Teams), Score Of(Current Array Element)))));
            If(Compare(Count Of(Filtered Array(All Players(All Teams), Compare(Score Of(Current Array Element), ==, Global.HighestScore))), >, 1));
                Set Global Variable(TieBreakerActive, True);
            End;
            Set Global Variable(HighestScore, Score Of(Last Of(Sorted Array(Filtered Array(All Players(All Teams), Not(And(Compare(Current Array Element, ==, Host Player), Value In Array(Global.rankedState, 21)))), Score Of(Current Array Element)))));
            If(Compare(Count Of(Filtered Array(Filtered Array(All Players(All Teams), Not(And(Compare(Current Array Element, ==, Host Player), Value In Array(Global.rankedState, 21)))), Compare(Score Of(Current Array Element), ==, Global.HighestScore))), >, 1));
                Set Global Variable(TieBreakerActive, True);
            End;
        End;
        Stop Chasing Global Variable(ballSpawnCountdown);
        Big Message(All Players(All Teams), Custom String("{0} has won the round", First Of(All Living Players(All Teams))));
        Wait(Value In Array(Global.roundTimings, 1), Ignore Condition);
        "only here, once.\nA press of the sit-out button during the win pause counts for the next round. If the host now sits out and\ntoo few players are left, no round: \"Check for <2 players\" stops the lobby (it may only see it next tick)"
        Call Subroutine(applyHostSitOut);
        "Tiebreaker or end of match, once per round end, among the players of the next round. If there is already a\ntiebreaker, it has ended. A match that ends here starts no round and brings nobody back."
        If(And(Or(Compare(Match Time, ==, 0), Global.TieBreakerActive), Not(Global.tournamentMode)));
            Call Subroutine(decideTiebreaker);
            Abort If(And(Compare(Match Time, ==, 0), Not(Global.TieBreakerActive)));
        End;
        "No round from here: too few players. The lobby stopped during the win pause, or is about to. Bring back the\nplayers while it waits; \"Wait for more players\" starts the next round. No tie is possible with one player."
        If(Compare(Subtract(Number Of Players(All Teams), If-Then-Else(And(Value In Array(Global.rankedState, 21), Array Contains(All Players(All Teams), Host Player)), 1, 0)), <=, 1));
            Set Global Variable(RoundSetupPending, False);
            Call Subroutine(resurrectForRound);
            Abort;
        End;
        "The lobby stopped during the win pause and enough players are back: start it again here, as \"Wait for more\nplayers\" would. In the same tick as bringing players back, so the lobby is never \"enough players\" with one alive."
        If(Not(Global.IsEnoughPlayersToStart));
            Big Message(All Players(All Teams), Custom String("Let's begin!"));
            Set Global Variable(IsEnoughPlayersToStart, True);
        End;
        If(Not(Global.TieBreakerActive));
            Call Subroutine(resurrectForRound);
        Else;
            "Kill anyone alive who doesn't have the highest score, and a host who sits out."
            Kill(Filtered Array(All Living Players(All Teams), Or(Compare(Score Of(Current Array Element), !=, Global.HighestScore), And(Compare(Current Array Element, ==, Host Player), Value In Array(Global.rankedState, 21)))), Null);
            "During tiebreaker, only revive the people who have the same score as the highest score, but a host who sits out."
            Resurrect(Filtered Array(All Dead Players(All Teams), And(Compare(Score Of(Current Array Element), ==, Global.HighestScore), Not(And(Compare(Current Array Element, ==, Host Player), Value In Array(Global.rankedState, 21))))));

rule ("Collision - collision check") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.ballIsOut == True;
        "RELIABLE_HITS is always false: this runs above FAST_BALL_SPEED only, as a backup for\n\"Collision - ball reaches player\", whose condition reads the chased ballPosition."
        Or(Compare(Global.ballSpeed, >, Value In Array(Global.ballFeel, 9)), Value In Array(Global.ballFeel, 10)) == True;
    }
    actions {
        If(Compare(Global.ballHitPosition, !=, Null));
            "-o--x--------o->Between last ballPosition & middlePosition"
            Set Global Variable At Index(phasePosition, 1, Add(Global.ballHitPosition, First Of(Global.phasePosition)));
            Modify Global Variable At Index(phasePosition, 1, Divide, 2);
            "-o-----x-----o->Between last & current ballPosition"
            Set Global Variable At Index(phasePosition, 0, Global.ballHitPosition);
            Set Global Variable(ballHitPosition, Null);
        Else;
            "-o--x--------o->Between last ballPosition & middlePosition"
            Set Global Variable At Index(phasePosition, 1, Add(First Of(Global.phasePosition), Global.ballPosition));
            Modify Global Variable At Index(phasePosition, 1, Divide, 2);
            Modify Global Variable At Index(phasePosition, 1, Add, First Of(Global.phasePosition));
            Modify Global Variable At Index(phasePosition, 1, Divide, 2);
            Modify Global Variable At Index(phasePosition, 0, Add, Global.ballPosition);
            Modify Global Variable At Index(phasePosition, 0, Divide, 2);
        End;
        "-o-----------x->"
        If(Compare(Distance Between(Global.ballPosition, Eye Position(Global.target)), <=, Value In Array(Global.ballFeel, 2)));
            Call Subroutine(collisionTarget);
        Else If(Compare(Distance Between(First Of(Global.phasePosition), Eye Position(Global.target)), <=, Value In Array(Global.ballFeel, 2)));
            Call Subroutine(collisionTarget);
        Else If(Compare(Distance Between(Value In Array(Global.phasePosition, 1), Eye Position(Global.target)), <=, Value In Array(Global.ballFeel, 2)));
            Call Subroutine(collisionTarget);
        Else If(Compare(Distance Between(Divide(Add(First Of(Global.phasePosition), Global.ballPosition), 2), Eye Position(Global.target)), <=, Value In Array(Global.ballFeel, 2)));
            Call Subroutine(collisionTarget);
        End;
        "Save last ballPosition for next loop"
        Set Global Variable At Index(phasePosition, 0, Evaluate Once(Global.ballPosition));
        Wait(0, Ignore Condition);
        Loop If Condition Is True;
    }
}

        End;

        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Has Spawned(Event Player) == True;
        (Event Player).rankTier == 0;
    }
    actions {
        "As in v1.3.2 RANKED: the name isn't known the moment a bot takes a slot"
        Wait(0.25, Ignore Condition);
        Set Player Variable(Event Player, rankTier, Subtract(Count Of(Global.rankTags), 1));
        While(And(Compare((Event Player).rankTier, >, 0), Not(Array Contains(Array Slice(Value In Array(Global.rankTags, (Event Player).rankTier), 3, Subtract(Count Of(Value In Array(Global.rankTags, (Event Player).rankTier)), 3)), Custom String("{0}", Event Player)))));

        End;
        If(Compare((Event Player).rankTier, ==, 0));
            Set Player Variable(Event Player, rankTier, -1);
    }
}
